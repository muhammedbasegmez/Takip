        function pressPin(digit) {
            if (pinInput.length < 4) {
                pinInput += digit;

                }
            }

        function updatePinDots() {
            for (let i = 0; i < 4; i++) {
                const dot = document.getElementById(`dot${i}`);
                if (i < pinInput.length) dot.classList.add('filled');
                else dot.classList.remove('filled');
            }
        }

        function verifyPin() {
            if (pinInput === state.pin) {
                document.getElementById('pinScreen').classList.remove('active');
                pinInput = '';
                updatePinDots();
            } else {
                alert('Hatalı PIN!');
                clearPin();
            }
        }

            pinInput = '';
            updatePinDots();

                if (newPin.length === 4 && !isNaN(newPin)) {
                    state.pin = newPin;

                    state.pin = null;
                    saveState();
                    alert('PIN koruması kaldırıldı.');

        }

        function filterTransactions() {
            const search = document.getElementById('searchInput').value.toLowerCase();
            const dateFilter = document.getElementById('dateFilter').value;
            const categoryFilter = document.getElementById('categoryFilter').value;

            const now = new Date();

            return state.transactions.filter(tx => {
                const txDate = new Date(tx.date);

                const matchesSearch = tx.desc.toLowerCase().includes(search) || tx.category.toLowerCase().includes(search);
                const matchesCategory = (categoryFilter === 'all') || (tx.category === categoryFilter);

                let matchesDate = true;
                if (dateFilter === 'thisMonth') {
                    matchesDate = txDate.getMonth() === now.getMonth() && txDate.getFullYear() === now.getFullYear();
                } else if (dateFilter === 'today') {
                    matchesDate = txDate.toDateString() === now.toDateString();
                } else if (dateFilter === 'thisWeek') {
                    const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
                    matchesDate = txDate >= weekAgo;
                } else if (dateFilter === 'last30') {
                    const thirtyDaysAgo = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
                    matchesDate = txDate >= thirtyDaysAgo;
                } else if (dateFilter === 'lastMonth') {
                    const lastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
                    matchesDate = txDate.getMonth() === lastMonth.getMonth() && txDate.getFullYear() === lastMonth.getFullYear();
                } else if (dateFilter === 'custom') {
                    const start = document.getElementById('startDate').value;
                    const end = document.getElementById('endDate').value;
                    if (start) matchesDate = matchesDate && (tx.date >= start);
                    if (end) matchesDate = matchesDate && (tx.date <= end);
                }

                return matchesSearch && matchesCategory && matchesDate;
            });
        }

        function renderTransactions() {
            const list = document.getElementById('txList');
            const filtered = filterTransactions();

            list.innerHTML = '';
            let totalInc = 0;
            let totalExp = 0;

            document.getElementById('txCount').innerText = `${filtered.length} İşlem`;

            if (filtered.length === 0) {
                list.innerHTML = '<div style="text-align: center; color: var(--text-muted); padding: 20px;">İşlem bulunamadı.</div>';
            } else {
                filtered.forEach(tx => {
                    if (tx.type === 'gelir') totalInc += tx.amount;
                    else totalExp += tx.amount;

                    const isIncome = tx.type === 'gelir';
                    const amountClass = isIncome ? 'stat-value green' : 'stat-value red';
                    const sign = isIncome ? '+' : '-';

                    list.innerHTML += `
                        <div class="tx-card">
                            <div class="tx-info">
                                <div class="tx-title">${tx.desc}</div>
                                <div class="tx-sub">
                                    <span>${tx.category}</span> • <span>${tx.date}</span>
                                </div>
                            </div>
                            <div class="tx-right">
                                <div class="tx-amount ${amountClass}">${sign}₺${tx.amount.toFixed(2)}</div>
                                <div class="tx-actions">
                                    <button class="action-btn" onclick="duplicateTx(${tx.id})" title="Klonla">📋</button>
                                    <button class="action-btn" onclick="deleteTx(${tx.id})" title="Sil">🗑️</button>
                                </div>
                            </div>
                        </div>
                    `;
                });
            }

            document.getElementById('totalIncome').innerText = `+₺${totalInc.toFixed(2)}`;
            document.getElementById('totalExpense').innerText = `-₺${totalExp.toFixed(2)}`;
            const balance = totalInc - totalExp;
            const netEl = document.getElementById('netBalance');
            netEl.innerText = `${balance >= 0 ? '' : '-'}₺${Math.abs(balance).toFixed(2)}`;
            netEl.style.color = balance >= 0 ? 'var(--accent-green)' : 'var(--accent-red)';
        }

        function checkRecurringStatus() {
            const currentMonthKey = new Date().toISOString().slice(0, 7);
            const isProcessed = state.processedMonths.includes(currentMonthKey);

            if (!isProcessed && state.recurring.length > 0) {
                document.getElementById('recurringBanner').style.display = 'flex';
            } else {
                document.getElementById('recurringBanner').style.display = 'none';
            }
        }

        function processRecurringForCurrentMonth() {
            const currentMonthKey = new Date().toISOString().slice(0, 7);
            const now = new Date();

            state.recurring.forEach(rec => {
                const dateStr = `${currentMonthKey}-${String(rec.day).padStart(2, '0')}`;
                state.transactions.unshift({
                    id: Date.now() + Math.random(),
                    type: rec.type,
                    amount: rec.amount,
                    category: rec.category,
                    date: dateStr,
                    desc: `[Sabit] ${rec.desc}`
                });
            });

            state.processedMonths.push(currentMonthKey);
            saveState();
            checkRecurringStatus();
            renderTransactions();
            alert('Sabit işlemleriniz bu ay için başarıyla aktarıldı!');
        }

        function renderAnalysis() {
            const container = document.getElementById('chartContainer');
            container.innerHTML = '';

            const expenses = filterTransactions().filter(t => t.type === 'gider');
            const categoryTotals = {};
            let totalExp = 0;

            expenses.forEach(t => {
                categoryTotals[t.category] = (categoryTotals[t.category] || 0) + t.amount;
                totalExp += t.amount;
            });

            if (totalExp === 0) {
                container.innerHTML = '<div style="text-align: center; color: var(--text-muted); padding: 20px;">Gider verisi bulunamadı.</div>';
                return;
            }

            Object.keys(categoryTotals).sort((a,b) => categoryTotals[b] - categoryTotals[a]).forEach(cat => {
                const amount = categoryTotals[cat];
                const percentage = ((amount / totalExp) * 100).toFixed(1);

                container.innerHTML += `
                    <div class="chart-bar-item">
                        <div class="chart-bar-header">
                            <span>${cat}</span>
                            <span>₺${amount.toFixed(2)} (%${percentage})</span>
                        </div>
                        <div class="chart-bar-track">
                            <div class="chart-bar-fill" style="width: ${percentage}%"></div>
                        </div>
                    </div>
                `;
            });
        }

            const name = prompt('Yeni Kategori Adı:');

                renderCategoryManager();

        function renameCategory(index) {
            const oldName = state.categories[index];
            const newName = prompt('Kategori Adını Düzenle:', oldName);
            if (newName && newName !== oldName) {
                state.categories[index] = newName;
                state.transactions.forEach(t => { if (t.category === oldName) t.category = newName; });
                state.recurring.forEach(r => { if (r.category === oldName) r.category = newName; });
                saveState();
                populateCategories();
                renderTransactions();
                renderCategoryManager();
            }
        }

        function deleteCategory(index) {
            if (confirm('Bu kategoriyi silmek istediğinizden emin misiniz?')) {
                state.categories.splice(index, 1);
                saveState();
                populateCategories();
                renderCategoryManager();
            }
        }

        function importJSON(event) {
            const fileReader = new FileReader();
            fileReader.onload = function (e) {
                try {
                    const importedState = JSON.parse(e.target.result);
                    if (importedState.transactions && importedState.categories) {
                        state = importedState;
                        saveState();
                        init();
                        alert('Yedek başarıyla yüklendi!');
                    } else {
                        alert('Geçersiz yedek dosyası.');
                    }
                } catch (err) {
                    alert('Hata: Dosya okunamadı.');
                }
            };
            fileReader.readAsText(event.target.files[0]);
        }
